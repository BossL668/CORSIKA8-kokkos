cmake_minimum_required(VERSION 3.16)
project(C8NativeMultiGpuTests LANGUAGES CXX)
include(GNUInstallDirs)
find_package(Arrow CONFIG REQUIRED)
find_package(yaml-cpp CONFIG REQUIRED)
find_package(CLI11 CONFIG REQUIRED)
add_subdirectory(../../../applications/detail/air_shower_multigpu coordinator)
add_executable(native_multigpu_cli ../../../applications/detail/air_shower_multigpu/NativeMain.cpp)
target_link_libraries(native_multigpu_cli PRIVATE c8_multigpu_coordinator)
add_executable(test_native_multigpu test_native_multigpu.cpp)
target_include_directories(test_native_multigpu PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../..)
target_link_libraries(test_native_multigpu PRIVATE c8_multigpu_coordinator)
add_executable(native_io_probe NativeIoProbe.cpp)
target_link_libraries(native_io_probe PRIVATE c8_multigpu_coordinator)
find_package(Threads REQUIRED)
add_executable(test_small_stack_digest test_small_stack_digest.cpp)
target_link_libraries(test_small_stack_digest PRIVATE c8_multigpu_coordinator Threads::Threads)
enable_testing()
add_test(NAME native_multigpu COMMAND test_native_multigpu $<TARGET_FILE:native_multigpu_cli>)
add_test(NAME small_stack_digest COMMAND test_small_stack_digest)
