cmake_minimum_required(VERSION 3.21)
project(InterfaceRadioConsumer LANGUAGES CXX CUDA)
find_package(Kokkos 4.7.3 CONFIG REQUIRED)
find_package(Eigen3 CONFIG REQUIRED)
find_package(CUDAToolkit REQUIRED)
if(NOT CORSIKA_BUILD OR NOT RADIO_EXECUTION MATCHES "^(CUDA|OPENMP)$")
  message(FATAL_ERROR "Set CORSIKA_BUILD to the exported build and RADIO_EXECUTION to CUDA or OPENMP")
endif()
include("${CORSIKA_BUILD}/corsikaTargets.cmake")
add_executable(radio_consumer main.cpp)
set_source_files_properties(main.cpp PROPERTIES LANGUAGE CUDA)
set_target_properties(radio_consumer PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES 75)
target_compile_definitions(radio_consumer PRIVATE RADIO_CONSUMER_${RADIO_EXECUTION}=1)
target_compile_options(radio_consumer PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-fopenmp>)
target_link_options(radio_consumer PRIVATE "LINKER:--no-as-needed")
# Deliberately no CORSIKA8::CORSIKA8, transport, air, PROPOSAL or shower target.
target_link_libraries(radio_consumer PRIVATE CORSIKA8::InterfaceRadio CUDA::cudart CUDA::cuda_driver)
