cmake_minimum_required(VERSION 3.21)
project(InterfaceCoreasConsumer LANGUAGES CXX CUDA)
find_package(Kokkos 4.7.3 CONFIG REQUIRED)
find_package(Eigen3 CONFIG REQUIRED)
find_package(CUDAToolkit REQUIRED)
if(NOT CORSIKA_BUILD OR NOT RADIO_EXECUTION MATCHES "^(CUDA|OPENMP)$")
  message(FATAL_ERROR "Set CORSIKA_BUILD to the exported build and RADIO_EXECUTION to CUDA or OPENMP")
endif()
include("${CORSIKA_BUILD}/corsikaTargets.cmake")
add_executable(coreas_fixture main.cpp)
add_executable(coreas_air_reference air_reference.cpp)
add_executable(coreas_pair_fixture paired.cpp)
set_source_files_properties(main.cpp air_reference.cpp paired.cpp PROPERTIES LANGUAGE CUDA)
foreach(c8_target coreas_fixture coreas_air_reference coreas_pair_fixture)
set_target_properties(${c8_target} PROPERTIES CUDA_STANDARD 17 CUDA_ARCHITECTURES 75)
target_compile_definitions(${c8_target} PRIVATE RADIO_CONSUMER_${RADIO_EXECUTION}=1)
target_compile_options(${c8_target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-fopenmp;--extended-lambda>)
target_link_options(${c8_target} PRIVATE "LINKER:--no-as-needed")
# Deliberately no CORSIKA8::CORSIKA8, transport, air, PROPOSAL or shower target.
target_link_libraries(${c8_target} PRIVATE CORSIKA8::InterfaceRadio CUDA::cudart CUDA::cuda_driver)
endforeach()
