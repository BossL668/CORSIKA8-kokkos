# Independently linkable material-interface transport. No air target links it.
set(C8_INTERFACE_TEST_THREADS "2" CACHE STRING "OpenMP threads for isolated material-interface tests")
set(C8_INTERFACE_EXECUTION_SPACE "DEFAULT" CACHE STRING "Single execution space for material-interface transport")
set_property(CACHE C8_INTERFACE_EXECUTION_SPACE PROPERTY STRINGS DEFAULT OPENMP CUDA)
if(NOT C8_INTERFACE_EXECUTION_SPACE MATCHES "^(DEFAULT|OPENMP|CUDA)$")
  message(FATAL_ERROR "C8_INTERFACE_EXECUTION_SPACE must be DEFAULT, OPENMP or CUDA")
endif()
add_library(CORSIKA8InterfaceEm STATIC InterfaceEmSession.cpp)
add_library(CORSIKA8::InterfaceEm ALIAS CORSIKA8InterfaceEm)
add_library(CORSIKA8TerrainEm ALIAS CORSIKA8InterfaceEm)
target_link_libraries(CORSIKA8InterfaceEm PUBLIC CORSIKA8ProposalNative Kokkos::kokkos CORSIKA8InterfaceRadio)
target_compile_definitions(CORSIKA8InterfaceEm PUBLIC CORSIKA8_WITH_KOKKOS=1)
target_compile_definitions(CORSIKA8InterfaceEm PUBLIC C8_INTERFACE_EXECUTION_${C8_INTERFACE_EXECUTION_SPACE}=1)
target_compile_features(CORSIKA8InterfaceEm PUBLIC cxx_std_17)
set_target_properties(CORSIKA8InterfaceEm PROPERTIES POSITION_INDEPENDENT_CODE ON EXPORT_NAME InterfaceEm)
if(CORSIKA_KOKKOS_BACKEND MATCHES "^CUDA(_OPENMP)?$")
  set_source_files_properties(InterfaceEmSession.cpp PROPERTIES LANGUAGE CUDA)
  target_link_libraries(CORSIKA8InterfaceEm PUBLIC CUDA::cudart CUDA::cuda_driver)
  target_compile_options(CORSIKA8InterfaceEm PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda>)
  set_target_properties(CORSIKA8InterfaceEm PROPERTIES
    CUDA_ARCHITECTURES "${CORSIKA_KOKKOS_CUDA_ARCHITECTURES}" CUDA_STANDARD 17)
  if(CORSIKA_KOKKOS_BACKEND STREQUAL "CUDA_OPENMP")
    target_compile_options(CORSIKA8InterfaceEm PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-fopenmp>)
  endif()
elseif(CORSIKA_KOKKOS_BACKEND STREQUAL "HIP")
  set_source_files_properties(InterfaceEmSession.cpp PROPERTIES LANGUAGE HIP)
endif()
install(TARGETS CORSIKA8InterfaceEm EXPORT CORSIKA8PublicTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
