name: Kokkos developer guide

on:
  pull_request:
    paths:
      - 'documentation/kokkos/**'
      - 'tools/build_kokkos_docs.py'
      - '.github/workflows/kokkos-docs.yml'
  push:
    paths:
      - 'documentation/kokkos/**'
      - 'tools/build_kokkos_docs.py'
      - '.github/workflows/kokkos-docs.yml'
  workflow_dispatch:
    inputs:
      publish:
        description: Publish the checked guide to GitHub Pages
        type: boolean
        default: false

permissions:
  contents: read

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.11'
          cache: pip
          cache-dependency-path: documentation/kokkos/requirements.txt
      - name: Install documentation dependencies
        run: python -m pip install -r documentation/kokkos/requirements.txt
      - name: Build and check internal links
        run: python tools/build_kokkos_docs.py
      - name: Save the preview
        uses: actions/upload-artifact@v4
        with:
          name: kokkos-developer-guide
          path: build/kokkos-docs
      - name: Prepare the Pages artifact
        if: github.event_name == 'workflow_dispatch' && inputs.publish && github.ref_name == github.event.repository.default_branch
        uses: actions/upload-pages-artifact@v4
        with:
          path: build/kokkos-docs/html

  deploy:
    if: github.event_name == 'workflow_dispatch' && inputs.publish && github.ref_name == github.event.repository.default_branch
    needs: build
    runs-on: ubuntu-latest
    permissions:
      pages: write
      id-token: write
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    concurrency:
      group: kokkos-guide-pages
      cancel-in-progress: false
    steps:
      - name: Configure Pages
        uses: actions/configure-pages@v5
      - name: Deploy the checked site
        id: deployment
        uses: actions/deploy-pages@v4
