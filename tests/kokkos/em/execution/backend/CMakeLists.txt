
set (BC_INSTALLED FALSE)
set (BC_NATIVE CORSIKA8)
set (BC_LIBRARY_PREFIX CORSIKA8)
set (BC_PREFIX production)
set (BC_BACKEND NONE)
if (TARGET CORSIKA8BackendRuntime)
  get_target_property (BC_RUNTIME_DEFINITIONS CORSIKA8BackendRuntime INTERFACE_COMPILE_DEFINITIONS)
  set (BC_BACKEND_DECLARATIONS)
  foreach (DEFINITION IN LISTS BC_RUNTIME_DEFINITIONS)
    if (DEFINITION MATCHES "^CORSIKA_KOKKOS_BACKEND_([A-Z_]+)(=1)?$")
      list (APPEND BC_BACKEND_DECLARATIONS "${CMAKE_MATCH_1}")
    endif ()
  endforeach ()
  list (LENGTH BC_BACKEND_DECLARATIONS BC_BACKEND_DECLARATION_COUNT)
  if (NOT BC_BACKEND_DECLARATION_COUNT EQUAL 1)
    message (FATAL_ERROR "BackendRuntime must publish exactly one backend capability")
  endif ()
  list (GET BC_BACKEND_DECLARATIONS 0 BC_BACKEND)
  if (NOT BC_BACKEND MATCHES "^(OPENMP|CUDA|CUDA_OPENMP|HIP|SYCL)$")
    message (FATAL_ERROR "Unknown BackendRuntime capability: ${BC_BACKEND}")
  endif ()
endif ()
set (BC_REAL_EXPORT "${CORSIKA_ENABLE_PROPOSAL_EXPORT}")
set (BC_BACKEND_TESTS "${CMAKE_CURRENT_SOURCE_DIR}")
set (BC_SESSION_TESTS "${CMAKE_CURRENT_SOURCE_DIR}/../session")
set (BC_SESSION_REFERENCE "${BC_SESSION_TESTS}/reference")
set (BC_FIXTURE_ROOT "${PROJECT_SOURCE_DIR}")
include ("${PROJECT_SOURCE_DIR}/cmake/BackendExecutionContracts.cmake")
